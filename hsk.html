        // --- AUDIO ENGINE ---

        let audioCtx;

        function initAudio() {

            if (!audioCtx) {

                audioCtx = new (window.AudioContext || window.webkitAudioContext)();

            }

            if (audioCtx.state === 'suspended') audioCtx.resume();

        }


        function playTone(frequency, type, duration, vol=0.1) {

            if (!state.soundOn || !audioCtx) return;

            const osc = audioCtx.createOscillator();

            const gain = audioCtx.createGain();

            osc.type = type;

            osc.frequency.setValueAtTime(frequency, audioCtx.currentTime);

            gain.gain.setValueAtTime(vol, audioCtx.currentTime);

            gain.gain.exponentialRampToValueAtTime(0.001, audioCtx.currentTime + duration);

            osc.connect(gain);

            gain.connect(audioCtx.destination);

            osc.start();

            osc.stop(audioCtx.currentTime + duration);

        }


        function playSound(type) {

            if (!state.soundOn) return;

            initAudio();

            if (type === 'click') playTone(600, 'sine', 0.1);

            else if (type === 'success') {

                playTone(523.25, 'sine', 0.1, 0.2); 

                setTimeout(() => playTone(659.25, 'sine', 0.1, 0.2), 100); 

                setTimeout(() => playTone(783.99, 'sine', 0.2, 0.2), 200); 

            } 

            else if (type === 'wrong') {

                playTone(300, 'sawtooth', 0.2, 0.2);

                setTimeout(() => playTone(250, 'sawtooth', 0.3, 0.2), 150);

            }

        }


        function speakCurrentSentence() {

            const text = state.currentQ.pre + state.currentQ.ans + state.currentQ.post;

            speakText(text);

        }


        function speakText(text) {

            if (!state.soundOn) return;

            if ('speechSynthesis' in window) {

                speechSynthesis.cancel();

                const utterance = new SpeechSynthesisUtterance(text);

                utterance.lang = 'zh-CN'; // Tiếng Trung

                utterance.rate = 0.8; 

                speechSynthesis.speak(utterance);

            }

        }

        // --- GAME FLOW ---

        function shuffleArray(arr) { return arr.sort(() => Math.random() - 0.5); }


        function startGameFlow() {

            initAudio();

            state.score = 0;

            state.lives = state.maxLives;

            state.correct = 0;

            state.wrong = 0;

            state.currentIndex = 0;

            

            // Lấy ngẫu nhiên 5 câu hỏi

            state.questionQueue = shuffleArray([...allQuestions]).slice(0, state.totalQuestions);

            

            showScreen('screen-game');

            loadNextQuestion();

        }


        function loadNextQuestion() {

            if(state.currentIndex >= state.totalQuestions || state.lives <= 0) {

                return showResult();

            }

            

            state.currentQ = state.questionQueue[state.currentIndex];

            state.currentIndex++;

            updateHUD();

            

            // Cập nhật câu hỏi lên giao diện

            document.getElementById('q-pre').innerText = state.currentQ.pre;

            document.getElementById('q-post').innerText = state.currentQ.post;

            

            const blank = document.getElementById('q-blank');

            blank.innerText = '';

            blank.style.borderBottomStyle = 'dashed';


            // Cập nhật nút lựa chọn

            const choicesArr = shuffleArray([...state.currentQ.choices]);

            const wordBank = document.getElementById('word-bank');

            wordBank.innerHTML = choicesArr.map(c => 

                `<div class="word-pill" onclick="checkAnswer('${c}')">${c}</div>`

            ).join('');

        }


        function updateHUD() {

            document.getElementById('hud-score').innerText = state.score;

            document.getElementById('hud-hearts').innerText = '❤️'.repeat(state.lives) + '🖤'.repeat(state.maxLives - state.lives);

            document.getElementById('hud-q-count').innerText = `${state.currentIndex}/${state.totalQuestions}`;

            

            const progress = ((state.currentIndex - 1) / state.totalQuestions) * 100;

            document.getElementById('hud-progress').style.width = `${progress}%`;

        }


        function checkAnswer(selectedWord) {

            const isCorrect = (selectedWord === state.currentQ.ans);

            const blank = document.getElementById('q-blank');

            blank.innerText = selectedWord;

            blank.style.borderBottomStyle = 'solid';


            if(isCorrect) {

                playSound('success');

                state.score += 20;

                state.correct++;

                showFeedback(true, state.currentQ.ans);

                speakCurrentSentence(); // Đọc nguyên câu khi đúng

            } else {

                playSound('wrong');

                state.lives--;

                state.wrong++;

                document.getElementById('q-box').classList.add('shake');

                setTimeout(() => document.getElementById('q-box').classList.remove('shake'), 500);

                

                if(state.lives <= 0) {

                    setTimeout(showResult, 1000);

                } else {

                    showFeedback(false, state.currentQ.ans);

                }

            }

        }

        // --- CONFETTI EFFECT ---

        function fireConfetti() {

            const canvas = document.getElementById('confetti');

            const ctx = canvas.getContext('2d');

            canvas.width = window.innerWidth;

            canvas.height = window.innerHeight;

            

            const particles = [];

            const colors = ['#f44336', '#e91e63', '#9c27b0', '#673ab7', '#3f51b5', '#2196f3', '#03a9f4', '#00bcd4', '#009688', '#4CAF50', '#8BC34A', '#CDDC39', '#FFEB3B', '#FFC107', '#FF9800', '#FF5722'];

            

            for(let i=0; i<100; i++) {

                particles.push({

                    x: canvas.width / 2,

                    y: canvas.height / 2 + 100,

                    r: Math.random() * 6 + 2,

                    dx: Math.random() * 10 - 5,

                    dy: Math.random() * -10 - 5,

                    color: colors[Math.floor(Math.random() * colors.length)]

                });

            }


            function animate() {

                requestAnimationFrame(animate);

                ctx.clearRect(0, 0, canvas.width, canvas.height);

                particles.forEach(p => {

                    ctx.beginPath();

                    ctx.arc(p.x, p.y, p.r, 0, Math.PI * 2);

                    ctx.fillStyle = p.color;

                    ctx.fill();

                    p.x += p.dx;

                    p.y += p.dy;

                    p.dy += 0.2; // Gravity

                });

            }

            animate();

            setTimeout(() => { ctx.clearRect(0,0,canvas.width, canvas.height); }, 3000);

        }
